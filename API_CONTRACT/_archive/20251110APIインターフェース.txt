/**
 * API Type Definitions
 * =====================
 * Shared types used across all API modules
 */

export interface Company {
  id: string;

  //企業情報
  nameJA: string;
  nameFurigana: string;
  nameEN: string;
  email: string;
  phoneNumber: string;
  corpNumber: string;
  address: string;
  fax?: string; //任意

  //代表者情報
  repFirstName: string;
  repLastName: string;
  repFirstNameKana: string;
  repLastNameKana: string;
  repEmail: string;
  repPhoneNumber: string;
  repPosition: string;

  //その他情報
  applicationDate: string;
}

//ユーザー情報
export interface User {
  id: string;
  username: string;
  email: string;
  role: "マスター" | "一般";
  companyId: string;
  siteId?: string;
}

//ユーザー詳細情報
export interface UserProfile extends User {
  firstName: string;
  lastName: string;
  firstNameKana: string;
  lastNameKana: string;
  phoneNumber: string;
  position: string;
  siteId: string;
}

export interface UserProfileResponse {
  statusCode: string;
  profile: UserProfile;
  company: Company; 
}

//ログインレスポンス
export interface LoginResponse {
  statusCode: string;
  token: string;
  user: User;
}

//会員登録データ
export interface SignUpData {
  corpNumber: string;
  companyNameJA: string;
  companyNameFurigana: string;
  companyNameEN: string;
  postalCode: string;
  prefectureID: string;
  cityID: string;
  addressLine: string;
  repFirstName: string;
  repLastName: string;
  repFirstNameKana: string;
  repLastNameKana: string;
  phoneNumber: string;
  repEmail: string;
  methodology: string[];
  equipment: Equipment[];
}

//会員登録レスポンス
export interface SignUpResponse {
  success: boolean;
  message: string;
  userId: string;
}

//設備情報
export interface Equipment {
  id: string;
  startDate: string;
  maker: string;
  model: string;
  unitCount: number;
  hasSusbsidy: boolean;
  subsidyName?: string;
  createdAt: string;
}

//プロジェクト情報
export interface Project {
  id: string;
  projectName: string;
  projectNumber: string;
  emissionReduction?: string;
  baselineEmissions?: string;
  methodCategory: string;
  methodNumber: string;
  methodName: string;
  methodMainEmission: string;

  //UI関連
  isJoined?: boolean;
  appliedDate?: string;
  status?: "active" | "maintenance" | "inactive" | "pending";
}

//参加企業とプロジェクトの関係
export interface CompanyProjectRelation {
  id: string;
  projectID: string;
  siteID: string;
}

//拠点情報
export interface Site {
  id: string;
  siteName: string;
  address: string;
  picFirstName: string;
  picLastName: string;
  picEmail: string;
  picPhoneNumber: string;
  picPosition: string;
  companyID: string;
}

//プロジェクト詳細情報 処理中
export interface ProjectDetail extends Project {
  companyName: string;
  site: string;
  deviceList: Array<{
    id: string;
    name: string;
    type: string;
    status: string;
    lastReading: string;
  }>;
  recentData: Array<{
    timestamp: string;
    parameter: string;
    value: number;
    unit: string;
  }>;
}

//モニタリングデータ 処理中
export interface MonitoringData {
  projectId: string;
  deviceId: string;
  parameter: string;
  value: number;
  unit: string;
  timestamp: string;
}

//データ送信レスポンス
export interface SubmitDataResponse {
  success: boolean;
  message: string;
  dataId: string;
}

//エラーレスポンス
export interface ErrorResponse {
  code?: string;
  message: string;
  statusCode?: number;
  details?: Record<string, any>;
  field?: string;
}
