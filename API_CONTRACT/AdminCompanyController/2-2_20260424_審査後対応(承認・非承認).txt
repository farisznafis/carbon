# 2-2 審査後対応(承認・非承認)
# approve/reject pending account
# This API is called when the Admin approves or rejects a pending company registration application. Upon approval, a master user account is created for the company. Upon rejection, the registration data is deleted.

POST /admin/companies/{companyId}/approve
POST /admin/companies/{companyId}/reject
Authorization: Bearer <token>
Content-Type: application/json

Request Body (JSON):
{
    "rejectionReason": null
}


Response - Success (HTTP 200)
{
  "statusCode": "OK",
  "message": "会社の承認が完了し、マスターユーザーが作成されました。",
  "companyId": "COMP001",
  "masterUsername": "user@example.com"
}

Response - Success (HTTP 200, reject)
{
  "statusCode": "OK",
  "message": "会社の申請は却下され、登録データは削除されました。",
  "companyId": "COMP001"
}

ERROR RESPONSES
Response - Error (HTTP 401)
{
  "statusCode": "UNAUTHORIZED",
  "message": "認証に失敗しました。有効なトークンを提供してください。"
}
Response - Error (HTTP 403)
{
  "statusCode": "FORBIDDEN",
  "message": "このリソースにアクセスする権限がありません。"
}
Response - Error (HTTP 404)
{
  "statusCode": "NOT_FOUND",
  "message": "該当する登録申請が見つかりませんでした。"
}
Response - Error (HTTP 500)
{
  "statusCode": "SERVER_ERROR",
  "message": "サーバー内部でエラーが発生しました。"
}
