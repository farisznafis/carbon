運営ユーザーのログインOTP検証
//POST /auth/admin/login が "statusCode": "OTP_REQUIRED" を返した場合の2段階目。
//メールで届いた6桁の認証コードを検証し、ログインを完了する。
//メールOTPログイン機能が有効な場合、運営ユーザーは全員が対象。
//ログイン不要（公開エンドポイント）。

URL: POST /auth/admin/login/email-otp/verify
Content-Type: application/json

Request Body (JSON):
{
	"challengeId": "7c9e6679-7425-40de-944b-e07fc1f90ae7", //ログイン応答の challengeId（UUID）
	"code": "123456" //メールで届いた6桁の数字
}

//参考: 1段階目 POST /auth/admin/login の応答（HTTP 200）
//{
//	"statusCode": "OTP_REQUIRED",
//	"message": "メール認証コードの入力が必要です。",
//	"challengeId": "7c9e6679-7425-40de-944b-e07fc1f90ae7",
//	"maskedEmail": "u***@example.com",
//	"expiresInSeconds": 300
//}

Response – Success (HTTP 200):
{
	"statusCode": "OK",
	"token": "eyJhbGciOiJIUzI1NiIsInR5cCI6...",
	"admin": {
		"id": null,
		"username": "tyamada",
		"name": "山田 太郎", //姓 + 半角スペース + 名
		"role": "operator",
		"email": "user@example.com",
		"firstName": "太郎",
		"firstNameKana": null,
		"lastName": "山田",
		"lastNameKana": null,
		"phoneNumber": "03-1234-5678"
	}
}

//認証コードの有効期限は5分。5回間違えるとチャレンジは破棄され、ログインからやり直しになる。
//admin の id / firstNameKana / lastNameKana はこのAPIでは設定されず、null で返る。

Response – Error:
(HTTP 401) //コード不一致・期限切れ・試行回数超過・使用済み、ユーザーまたは所属企業が無効、発行後にパスワード等が変更された
{
	"statusCode": "INVALID_OR_EXPIRED_TOKEN",
	"message": "セキュリティトークンが無効か、有効期限が切れています。"
}
--
(HTTP 400)
{
	"statusCode": "VALIDATION_ERROR",
	"message": "入力内容に誤りがあります。",
	"errors": {
		"code": "正規表現 \"\\d{6}\" にマッチさせてください" //標準メッセージ（Accept-Language により変わる）
	}
}
--
(HTTP 400) //challengeId が UUID 形式でない場合
{
	"statusCode": "BAD_REQUEST",
	"message": "入力値に誤りがあります。",
	"errors": {
		"challengeId": "入力形式が正しくありません。"
	}
}
--
(HTTP 429) //同一IPから15分間に20回を超えた場合（セキュリティ系POSTの合計）
{
	"statusCode": "TOO_MANY_REQUESTS",
	"message": "リクエストが多すぎます。しばらくしてから再度お試しください。"
}
--
(HTTP 500)
{
	"statusCode": "INTERNAL_SERVER_ERROR",
	"message": "サイト情報の取得中にエラーが発生しました。"
}
