招待によるアカウント作成
//招待リンクを開いた本人が、自分で情報を入力してアカウントを作成する。
//事前に「アクションリンク検証」で purpose = USER_INVITATION のアクションセッションを取得しておく。
//ログイン不要（公開エンドポイント）。

URL: POST /auth/invitation/complete
Content-Type: application/json
X-Action-Session: e81f4d7c2a9b6e3f8c1d5a7b9e2f4c6d

Request Body (JSON): //メールアドレスは招待時のものを使用するため、リクエストには含めない
{
	"username": "y.taro", //3〜50文字。半角英数字、スペース、@、_、絵文字
	"firstName": "太郎", //全角、50文字以内
	"firstNameKana": "タロウ", //カタカナ、50文字以内
	"lastName": "山田", //全角、50文字以内
	"lastNameKana": "ヤマダ", //カタカナ、50文字以内
	"phoneNumber": "03-1234-5678", //7〜16文字
	"userPosition": "担当者", //全角、50文字以内
	"password": "Abcd1234" //8〜64文字。大文字・小文字・数字を各1文字以上
}

Response – Success (HTTP 200):
{
	"statusCode": "OK",
	"message": "アカウントを作成しました。ログインしてください。"
}

//作成されるユーザーの role は "user"、拠点は招待時に指定されたもの。
//リンクとセッションは使用済みになる。

Response – Error:
(HTTP 401) //セッション不正・期限切れ（1時間）・purpose 不一致・使用済み、企業が有効でない
{
	"statusCode": "INVALID_OR_EXPIRED_TOKEN",
	"message": "セキュリティトークンが無効か、有効期限が切れています。"
}
--
(HTTP 409) //招待後に同じアドレスで登録された場合
{
	"statusCode": "USED_EMAIL",
	"message": "このメールアドレスは既に使用されています。"
}
--
(HTTP 409)
{
	"statusCode": "USED_USERNAME",
	"message": "このユーザ名は既に使用されています。"
}
--
(HTTP 409) //招待後に上限に達した場合（企業あたり50人、拠点あたり5人）
{
	"statusCode": "LIMIT_EXCEEDED",
	"message": "上限に達しています。"
}
--
(HTTP 400)
{
	"statusCode": "VALIDATION_ERROR",
	"message": "入力内容に誤りがあります。",
	"errors": {
		"username": "ユーザー名を入力してください。",
		"firstName": "名前（名）を入力してください。",
		"firstNameKana": "名前カナ（名）は50文字以内で入力してください。",
		"lastName": "名前（姓）を入力してください。",
		"lastNameKana": "名前カナ（姓）を入力してください。",
		"phoneNumber": "電話番号は7文字以上16文字以内で入力してください。",
		"userPosition": "担当者の役職を入力してください。",
		"password": "パスワードは8〜64文字で、大文字・小文字・数字をそれぞれ1文字以上含めてください。よく使われるパスワードは使用できません。"
	}
}
--
(HTTP 400) //X-Action-Session ヘッダーがない場合
{
	"statusCode": "BAD_REQUEST",
	"message": "Required request header 'X-Action-Session' for method parameter type String is not present"
}
--
(HTTP 404) //招待時の拠点が削除されている場合
{
	"statusCode": "NOT_FOUND",
	"message": "該当するサイトが見つかりませんでした。"
}
--
(HTTP 404) //機能が無効の場合
{
	"statusCode": "NOT_FOUND",
	"message": "該当するリソースが見つかりませんでした。"
}
--
(HTTP 429) //同一IPから15分間に20回を超えた場合（セキュリティ系POSTの合計）
{
	"statusCode": "TOO_MANY_REQUESTS",
	"message": "リクエストが多すぎます。しばらくしてから再度お試しください。"
}
--
(HTTP 500)
{
	"statusCode": "INTERNAL_SERVER_ERROR",
	"message": "サイト情報の取得中にエラーが発生しました。"
}
