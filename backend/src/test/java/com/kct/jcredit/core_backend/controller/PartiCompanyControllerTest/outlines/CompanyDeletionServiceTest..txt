    @Nested
    @DisplayName("approveDeletionRequest（POST /company/{companyId}/deletion-approve）のテスト")
    class ApproveDeletionRequestTest {

        @Test
        @DisplayName("UT-CDS-006: companyIdに紐づく削除申請が存在しない場合、REQUEST_NOT_FOUND例外をthrowする")
        void approveDeletionRequest_requestNotFound_throwsRequestNotFound() {
            // Input: companyId="COMP001"
            //        deletionRequestRepository.findByCompanyIdAndStatus("COMP001", PENDING)がOptional.emptyを返すようモック
            // Output: CustomException(400, REQUEST_NOT_FOUND, "削除申請が見つかりません。")がthrowされること
            //         companyRepository.saveが呼ばれないこと
        }

        @Test
        @DisplayName("UT-CDS-007: 削除申請のステータスがPENDING以外（APPROVED/REJECTED）の場合、INVALID_STATE例外をthrowする")
        void approveDeletionRequest_notPending_throwsInvalidState() {
            // Input: companyId="COMP001"
            //        deletionRequestRepository.findByCompanyIdAndStatus("COMP001", PENDING)が既にstatus=APPROVEDのリクエストを返すようモック（またはfindByIdでstatusチェックする実装の場合はそれに合わせる）
            // Output: CustomException(400, INVALID_STATE, "PENDING以外は処理できません。")がthrowされること
            //         companyRepository.saveが呼ばれないこと
        }

        @Test
        @DisplayName("UT-CDS-008: 削除申請はPENDINGだが紐づく企業が存在しない場合、COMPANY_NOT_FOUND例外をthrowする")
        void approveDeletionRequest_companyNotFound_throwsCompanyNotFound() {
            // Input: companyId="COMP001"
            //        deletionRequestRepository.findByCompanyIdAndStatus("COMP001", PENDING)がstatus=PENDINGのリクエストを返すようモック
            //        companyRepository.findById("COMP001")がOptional.emptyを返すようモック
            // Output: CustomException(400, COMPANY_NOT_FOUND, "企業が見つかりません。")がthrowされること
            //         companyRepository.saveが呼ばれないこと
        }

        @Test
        @DisplayName("UT-CDS-009: 正常に承認された場合、企業と関連データがカスケード論理削除され申請がAPPROVEDに更新される")
        void approveDeletionRequest_valid_cascadesSoftDeleteAndApprovesRequest() {
            // Input: companyId="COMP001"（Authorization: operator_adminのJWT）
            //        deletionRequestRepository.findByCompanyIdAndStatus("COMP001", PENDING)がrequestId=1, requestedBy="master_user", requestedAt=2026-06-19T10:30:00, reason="事業都合により脱退"のリクエストを返すようモック
            //        companyRepository.findById("COMP001")がcompany(companyName="株式会社サンプル")を返すようモック
            //        userRepository.findByCompanyId、siteRepository.findByCompanyId、projectRelationRepository等が関連エンティティ一覧を返すようモック
            // Output: statusCode=200, message="Deletion request approved and company soft-deleted successfully"
            //         data.requestId=1, companyId="COMP001", status="APPROVED", decidedBy="operator_admin", decidedAt=2026-06-19T11:00:00, rejectionReason=null
            //         companyRepository.saveに渡されたcompanyが論理削除状態（deletedAt設定、isActive=false等）であることを検証
            //         userRepository.saveAll／siteRepository.saveAll／projectRelationRepository.saveAll／equipmentRepository.saveAll／reductionResultRepository.saveAll等、関連エンティティが論理削除（deletedAt設定）されたことを検証
            //         deletionRequestRepository.saveに渡されたrequestのstatus=APPROVED、decidedBy、decidedAtが設定されていることを検証
        }

        @Test
        @DisplayName("UT-CDS-010: カスケード削除処理中に例外が発生した場合、トランザクションがロールバックされ一部のみ更新されないこと")
        void approveDeletionRequest_cascadeFailure_rollsBackTransaction() {
            // Input: companyId="COMP001"
            //        deletionRequestRepository.findByCompanyIdAndStatus("COMP001", PENDING)がstatus=PENDINGのリクエストを返すようモック
            //        companyRepository.findById("COMP001")がcompanyを返すようモック
            //        equipmentRepository.saveAll等、カスケード処理の一部でRuntimeExceptionをthrowするようモック
            // Output: 例外がそのままthrowされること
            //         companyRepository.saveで更新された内容がコミットされない（ロールバックされる）ことを検証（@Transactional前提のテスト、実際はトランザクション境界の確認）
            //         deletionRequestRepository.saveがAPPROVEDで呼ばれていないことを検証
        }

        @Test
        @DisplayName("UT-CDS-011: operator権限がないユーザがアクセスした場合、403 FORBIDDENをthrowする")

            // Input: companyId="COMP001"（Authorization: operator以外のロールのJWT）
            // Output: statusCode=403, message="FORBIDDEN"がthrowされること
            //         deletionRequestRepository.findByCompanyIdAndStatus、companyRepository.saveが呼ばれないこと
        }
    }
